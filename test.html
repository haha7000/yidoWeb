${setting.company_name} ${setting.branch_name} ${new Date(setting.effective_from).toLocaleDateString('ko-KR')} ${endDate} ${freeRatePercent}% 이상 수수료 면제 ${uploadDate}
`; listContainer.innerHTML = html; } // 면세점별 뱃지 클래스 function getStoreClass(companyName) { switch(companyName) { case 'LOTTE': return 'bg-warning'; case 'SHILLA': return 'bg-info'; case 'HYUNDAI': return 'bg-success'; default: return 'bg-secondary'; } } // 데이터 없음 메시지 표시 function showNoDataMessage() { const noDataMessage = document.getElementById('noDataMessage'); noDataMessage.style.display = 'block'; } // 수수료 상세 정보 보기 async function viewFeeDetail(settingsId, title) { const modal = new bootstrap.Modal(document.getElementById('feeDetailModal')); const modalTitle = document.getElementById('feeDetailModalLabel'); const modalContent = document.getElementById('feeDetailContent'); modalTitle.innerHTML = `${title} 수수료 상세 정보`; modalContent.innerHTML = `

수수료 정보를 불러오는 중...

`; modal.show(); try { const response = await fetch(`http://localhost:8000/api/fees/detail/${settingsId}`); if (response.ok) { const result = await response.json(); displayFeeDetailContent(result); } else { modalContent.innerHTML = `
수수료 정보를 불러오는데 실패했습니다.
`; } } catch (error) { console.error('수수료 상세 조회 오류:', error); modalContent.innerHTML = `
네트워크 오류가 발생했습니다.
`; } } // 수수료 상세 내용 표시 function displayFeeDetailContent(data) { const modalContent = document.getElementById('feeDetailContent'); let html = `

지점: ${data.settings.branch_name}

${data.settings.note ? `

` : ''}
`; // 카테고리별 수수료 탭 html += `
${generateFeeTable(data.category_fees, 'category')}
`; // 브랜드별 수수료 탭 html += `
${generateFeeTable(data.brand_fees, 'brand')}
`; // 품목별 수수료 탭 html += `
${generateFeeTable(data.item_fees, 'item')}
`; html += `
`; modalContent.innerHTML = html; } // 수수료 테이블 생성 function generateFeeTable(fees, type) { if (!fees || fees.length === 0) { return `
등록된 ${getTypeLabel(type)} 수수료가 없습니다
`; } let headers = ''; let rows = ''; if (type === 'category') { headers = ` 카테고리 코드 카테고리명 수입/로컬 수수료율 `; fees.forEach(fee => { rows += ` ${fee.category_code} ${fee.category_name || '-'} ${fee.import_type || '-'} ${(fee.commission_rate * 100).toFixed(2)}% `; }); } else if (type === 'brand') { headers = ` 카테고리 브랜드명 수수료율 `; fees.forEach(fee => { rows += ` ${fee.category_code} ${fee.brand_name} ${(fee.commission_rate * 100).toFixed(2)}% `; }); } else if (type === 'item') { headers = ` 카테고리 브랜드명 상품명 상품코드 수수료율 `; fees.forEach(fee => { rows += ` ${fee.category_code} ${fee.brand_name} ${fee.item_name || '-'} ${fee.product_code} ${(fee.commission_rate * 100).toFixed(2)}% `; }); } return `
${headers}${rows}
`; } // 타입별 라벨 function getTypeLabel(type) { switch(type) { case 'category': return '카테고리별'; case 'brand': return '브랜드별'; case 'item': return '품목별'; default: return ''; } } // 수수료 설정 다운로드 async function downloadFeeSetting(settingsId) { try { const response = await fetch(`http://localhost:8000/api/fees/export/${settingsId}`); if (response.ok) { const blob = await response.blob(); const url = window.URL.createObjectURL(blob); const a = document.createElement('a'); a.href = url; a.download = `수수료설정_${settingsId}.xlsx`; document.body.appendChild(a); a.click(); window.URL.revokeObjectURL(url); document.body.removeChild(a); showNotification('수수료 설정이 다운로드되었습니다.', 'success'); } else { showNotification('다운로드에 실패했습니다.', 'error'); } } catch (error) { console.error('다운로드 오류:', error); showNotification('다운로드 중 오류가 발생했습니다.', 'error'); } } // 수수료 설정 삭제 async function deleteFeeSetting(settingsId, title) { if (!confirm(`"${title}" 수수료 설정을 삭제하시겠습니까?\n이 작업은 되돌릴 수 없습니다.`)) { return; } try { const response = await fetch(`http://localhost:8000/api/fees/settings/${settingsId}`, { method: 'DELETE' }); if (response.ok) { showNotification('수수료 설정이 삭제되었습니다.', 'success'); // 목록 새로고침 loadUploadedFeeData(); } else { showNotification('삭제에 실패했습니다.', 'error'); } } catch (error) { console.error('삭제 오류:', error); showNotification('삭제 중 오류가 발생했습니다.', 'error'); } } // 목록 새로고침 function refreshUploadedData() { loadUploadedFeeData(); showNotification('목록을 새로고침했습니다.', 'info'); }
